# ชนิดจากโมดูล



ชนิดจากโมดูลหมายถึงความสามารถในการใช้ค่าที่โมดูลส่งออกเพื่ออนุมานชนิดของค่าเหล่านั้นโดยอัตโนมัติ เมื่อโมดูลส่งออกค่าที่มีชนิดเฉพาะ TypeScript สามารถใช้ข้อมูลนั้นเพื่ออนุมานชนิดของค่าโดยอัตโนมัติเมื่อนำเข้าไปยังโมดูลอื่น

<!-- skip -->
```typescript
// calc.ts
export const add = (x: number, y: number) => x + y;
// index.ts
import { add } from 'calc';
const r = add(1, 2); // r is number
```
